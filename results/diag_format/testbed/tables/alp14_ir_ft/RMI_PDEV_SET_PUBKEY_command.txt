B4.3.19 RMI_PDEV_SET_PUBKEY command
                Provide public key associated with a PDEV.
                B4.3.19.1 Interface
                B4.3.19.1.1   Input values
        Name                  Register    Bits        Type                                  Description
        fid                   X0          63:0        UInt64                                FID, value 0xC400017B
        pdev_ptr              X1          63:0        Address                               PA of the PDEV
        params_ptr            X2          63:0        Address                               PA of the key parameters
                B4.3.19.1.2   Context
                The RMI_PDEV_SET_PUBKEY command operates on the following context.
        Name                  Type                       Value                                      Before   Description
        pdev                  RmmPdev                    PdevAt(pdev_ptr)                           false    PDEV
        params                RmiPublicKeyParams         RmiPublicKeyParamsAt(                      false    Public key parameters
                                              params_ptr)
                B4.3.19.1.3   Output values
        Name                  Register    Bits        Type                                  Description
        result                X0          63:0        RmiCommandReturnCode                  Command return status
                B4.3.19.2 Failure conditions
  ID                           Condition
  da_supported                pre: false
                               post: ResultEqual(result, RMI_ERROR_NOT_SUPPORTED)
  pdev_ptr_align              pre: !AddrIsGranuleAligned(pdev_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  pdev_ptr_delegable          pre: !PaIsDelegable(pdev_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  pdev_ptr_state              pre: GranuleAt(pdev_ptr).state != PDEV
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  params_ptr_align            pre: !AddrIsGranuleAligned(params_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  params_ptr_access           pre: !AddrIsWithin(params_ptr, 0, 0)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  key_length                  pre: RmiPublicKeyParamsAt(params_ptr).key.len > 1024
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  metadata_length             pre: RmiPublicKeyParamsAt(params_ptr).metadata.len > 1024
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  key_valid                   pre: false
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  metadata_valid              pre: false
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  pdev_state                  pre: PdevAt(pdev_ptr).state != PDEV_NEEDS_KEY
                               post: ResultEqual(result, RMI_ERROR_DEVICE)
                B4.3.19.2.1   Failure condition ordering
                [da_supported, pdev_ptr_align, pdev_ptr_delegable, pdev_ptr_state, params_ptr_align, params_ptr_access, key_length, key_valid, metadata_length, metadata_valid] < [pdev_state]
                                                                                           da_supported                     pdev_ptr_align                     pdev_ptr_delegable                     pdev_ptr_state                     params_ptr_align                     params_ptr_access                     key_length                     key_valid                     metadata_length                     metadata_valid

                                                                                                                           RMI_ERROR_INPUT
                                                                                                                           RMI_ERROR_INPUT
                                                                                                                           RMI_ERROR_INPUT
                                                                                                                           RMI_ERROR_INPUT
                                                                                                                           RMI_ERROR_INPUT
                                                                                                                           pdev_state
                B4.3.19.3 Success conditions
  pdev_state                  post: PdevAt(pdev_ptr).state == PDEV_HAS_KEY
  pdev_comm_state             post: PdevAt(pdev_ptr).comm_state == DEV_COMM_PENDING
