B4.3.38 RMI_RTT_CREATE command
                Creates a primary RTT.
                B4.3.38.1 Interface
                B4.3.38.1.1   Input values
        Name                  Register    Bits        Type                                  Description
        fid                   X0          63:0        UInt64                                FID, value 0xC400015D
        rd                    X1          63:0        Address                               PA of the RD for the target Realm
        rtt                   X2          63:0        Address                               PA of the target RTT
        ipa                   X3          63:0        Address                               Base of the IPA range described by the RTT
        level                 X4          63:0        Int64                                 RTT level
                B4.3.38.1.2   Context
                The RMI_RTT_CREATE command operates on the following context.
        Name                  Type                       Value                                      Before   Description
        realm                 RmmRealm                   RealmAt(rd)                                false    Realm
        walk                  RmmRttWalkResult           RttWalk(                                   false    RTT walk result
                                                             realm, ipa,
                                                             level - 1,
                                                             RMM_RTT_TREE_PRIMARY)
        entry_idx             UInt64                     RttEntryIndex(                             false    RTTE index
                                                             ipa, walk.level)
        walk_pre              RmmRttWalkResult           RttWalk(                                   true     RTT walk result
                                                             realm, ipa,                                     before command
                                                             level - 1,                                      execution
                                                             RMM_RTT_TREE_PRIMARY)
        rtte_pre              RmmRttEntry                walk_pre.rtte                              true     RTTE before
                                                                                                             command execution
                B4.3.38.1.3   Output values
        Name                  Register    Bits        Type                                  Description
        result                X0          63:0        RmiCommandReturnCode                  Command return status
                B4.3.38.2 Failure conditions
  ID                           Condition
  rd_align                     pre: !AddrIsGranuleAligned(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_delegable                 pre: !PaIsDelegable(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_state                     pre: GranuleAt(rd).state != RD
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  level_valid                  pre: !(RttLevelIsValid(RealmAt(rd), level) && !(level == RealmAt(rd).rtt_level_start))
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  ipa_aligned                  pre: !AddrIsRttLevelAligned(ipa, level - 1)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  ipa_in_range                 pre: !(ipa < pow2(RealmAt(rd).ipa_width))
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rtt_aligned                  pre: !AddrIsGranuleAligned(rtt)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rtt_delegable                pre: !PaIsDelegableDram(rtt)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rtt_state                    pre: GranuleAt(rtt).state != DELEGATED
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  realm_lpa2                   pre: RealmAt(rd).feat_lpa2 == FEATURE_FALSE && rtt >= pow2(48)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  walk_below                   pre: RttWalk(RealmAt(rd), ipa, level - 1, RMM_RTT_TREE_PRIMARY).level < level - 1
                               post: ResultEqual(result, RMI_ERROR_RTT(0))
  rtte_table                   pre: RttEntryAt(RttAt(walk_pre.rtt_addr), RttEntryIndex(ipa, walk_pre.level)).state == TABLE
                               post: ResultEqual(result, RMI_ERROR_RTT(0))
                B4.3.38.2.1   Failure condition ordering
                [rd_delegable, rd_state, level_valid, ipa_in_range] < [walk_below, rtte_table]
  rd_delegable                 rd_state                        level_valid                     ipa_in_range
                                                                                           RMI_ERROR_INPUT
                                                                                           walk_below
                                                                                           rtte_table
                                                                                           RMI_ERROR_RTT
                B4.3.38.3 Success conditions
  rtt_state                    post: GranuleAt(rtt).state == RTT
  rtte_table                   post: RttEntryAt(RttAt(walk.rtt_addr), entry_idx).state == TABLE
  rtte_rtt                     post: RttEntryAt(RttAt(walk.rtt_addr), entry_idx).rtt == rtt
  rtte_ripas                   post: RttEntryAt(RttAt(walk.rtt_addr), entry_idx).ripas == rtte_pre.ripas
  rtte_state                   post: RttEntryAt(RttAt(walk.rtt_addr), entry_idx).state == rtte_pre.state
                B4.3.38.4 Footprint
    ID                        Value
    rtt_state                 GranuleAt(rtt).state
    rtte                      RttEntryAt(RttAt(walk.rtt_addr), entry_idx)
