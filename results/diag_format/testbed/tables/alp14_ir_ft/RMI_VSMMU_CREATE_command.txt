B4.3.65 RMI_VSMMU_CREATE command
              Create a VSMMU.
              B4.3.65.1 Interface
              B4.3.65.1.1   Input values
        Name                Register    Bits        Type                                  Description
        fid                 X0          63:0        UInt64                                FID, value 0xC400016A
        rd                  X1          63:0        Address                               PA of the RD
        vsmmu_ptr           X2          63:0        Address                               PA of the VSMMU
        params_ptr          X3          63:0        Address                               PA of VSMMU parameters
                B4.3.65.1.2   Context
              The RMI_VSMMU_CREATE command operates on the following context.
        Name                Type                       Value                                      Before   Description
        realm_pre           RmmRealm                   RealmAt(rd)                                true     Realm
        realm               RmmRealm                   RealmAt(rd)                                false    Realm
        vsmmu               RmmVsmmu                   VsmmuAt(vsmmu_ptr)                         false    VSMMU
        params              RmiVsmmuParams             RmiVsmmuParamsAt(                          false    VSMMU parameters
                                                           params_ptr)
                B4.3.65.1.3   Output values
        Name                Register    Bits        Type                                  Description
        result              X0          63:0        RmiCommandReturnCode                  Command return status
                B4.3.65.2 Failure conditions
  ID                           Condition
  feat_da                     pre: !ImplFeatures(s).feat_da
                               post: ResultEqual(result, RMI_ERROR_NOT_SUPPORTED)
  rd_align                    pre: !AddrIsGranuleAligned(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_delegable                pre: !PaIsDelegable(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_state                    pre: GranuleAt(rd).state != RD
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  realm_state                 pre: RealmAt(rd).state != REALM_NEW
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  vsmmu_ptr_align             pre: !AddrIsGranuleAligned(vsmmu_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  vsmmu_ptr_delegable         pre: !PaIsDelegable(vsmmu_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  vsmmu_ptr_state             pre: GranuleAt(vsmmu_ptr).state != DELEGATED
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  params_ptr_align            pre: !AddrIsGranuleAligned(params_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  params_ptr_access           pre: !GranuleAccessPermitted(params_ptr, PAS_NS)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  params_ptr_valid            pre: !RmiVsmmuParamsIsValid(params_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  reg_base_align              pre: !AddrIsGranuleAligned(RmiVsmmuParamsAt(params_ptr).reg_base)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  reg_top_align               pre: !AddrIsGranuleAligned(RmiVsmmuParamsAt(params_ptr).reg_top)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  reg_base_prot               pre: !AddrIsProtected(RmiVsmmuParamsAt(params_ptr).reg_base, RealmAt(rd))
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  reg_top_gt_reg_base         pre: RmiVsmmuParamsAt(params_ptr).reg_top <= RmiVsmmuParamsAt(params_ptr).reg_base
                               post: ResultEqual(result, RMI_ERROR_INPUT)
