B4.3.1 RMI_DATA_CREATE command
                Creates a Data Granule, copying contents from a Non-secure Granule provided by the caller.
                B4.3.1.1 Interface
                B4.3.1.1.1   Input values
        Name                  Register    Bits        Type                                  Description
        fid                   X0          63:0        UInt64                                FID, value 0xC4000153
        rd                    X1          63:0        Address                               PA of the RD for the target Realm
        data                  X2          63:0        Address                               PA of the target Data
        ipa                   X3          63:0        Address                               IPA at which the Granule will be
                                                                                              mapped in the target Realm
        src                   X4          63:0        Address                               PA of the source Granule
        flags                 X5          63:0        RmiDataFlags                          Flags
                B4.3.1.1.2   Context
                The RMI_DATA_CREATE command operates on the following context.
        Name                  Type                       Value                                      Before   Description
        realm_pre             RmmRealm                   RealmAt(rd)                                true     Realm
        realm                 RmmRealm                   RealmAt(rd)                                false    Realm
        walk                  RmmRttWalkResult            RttWalk(                                   false   RTT walk result
                                           realm, ipa,
                                           RMM_RTT_PAGE_LEVEL,
                                           RMM_RTT_TREE_PRIMARY)
        entry_idx             UInt64                      RttEntryIndex(                             false   RTTE index
                                           ipa, walk.level)
                B4.3.1.1.3   Output values
        Name                  Register    Bits        Type                                  Description
        result                X0          63:0        RmiCommandReturnCode                  Command return status
                B4.3.1.2 Failure conditions
  ID                           Condition
  src_align                    pre: !AddrIsGranuleAligned(src)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  src_access                   pre: !GranuleAccessPermitted(src, PAS_NS)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  data_align                   pre: !AddrIsGranuleAligned(data)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  data_delegable               pre: !PaIsDelegable(data)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  data_state                   pre: GranuleAt(data).state != DELEGATED
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  data_lpa2                    pre: data >= 2^48 && RealmAt(rd).feat_lpa2 == FEATURE_FALSE
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_align                     pre: !AddrIsGranuleAligned(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_delegable                 pre: !PaIsDelegable(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_state                     pre: GranuleAt(rd).state != RD
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  ipa_align                    pre: !AddrIsGranuleAligned(ipa)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  ipa_protected                pre: !AddrIsProtected(ipa, RealmAt(rd))
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  realm_state                  pre: RealmAt(rd).state != REALM_NEW
                               post: ResultEqual(result, RMI_ERROR_REALM(0))
  rtt_walk                     pre: RttWalk(RealmAt(rd), ipa,RMM_RTT_PAGE_LEVEL,RMM_RTT_TREE_PRIMARY).level < RMM_RTT_PAGE_LEVEL
                               post: ResultEqual(result, RMI_ERROR_RTT(RttWalk(RealmAt(rd), ipa,RMM_RTT_PAGE_LEVEL,RMM_RTT_TREE_PRIMARY).level))
  rtt_entry_state              pre: RttEntryAt(RttAt(RttWalk(RealmAt(rd), ipa,RMM_RTT_PAGE_LEVEL,RMM_RTT_TREE_PRIMARY).rtt_addr), RttEntryIndex(ipa, RttWalk(RealmAt(rd), ipa,RMM_RTT_PAGE_LEVEL,RMM_RTT_TREE_PRIMARY).level)).state != UNASSIGNED
                               post: ResultEqual(result, RMI_ERROR_RTT(RttWalk(RealmAt(rd), ipa,RMM_RTT_PAGE_LEVEL,RMM_RTT_TREE_PRIMARY).level))
                B4.3.1.2.1   Failure condition ordering
                [rd_delegable, rd_state] < [realm_state]
                [rd_delegable, rd_state] < [rtt_walk, rtt_entry_state]
                [ipa_protected] < [rtt_walk, rtt_entry_state]
                                 rd_delegable                     rd_state                        realm_state
                                                                                           RMI_ERROR_INPUT
                                                                                           rtt_walk
                                                                                           rtt_entry_state
                B4.3.1.3 Success conditions
  data_state                   post: GranuleAt(data).state == DATA
  rtte                         post: RttEntryAt(RttAt(RttWalk(RealmAt(rd), ipa,RMM_RTT_PAGE_LEVEL,RMM_RTT_TREE_PRIMARY).rtt_addr), RttEntryIndex(ipa, RttWalk(RealmAt(rd), ipa,RMM_RTT_PAGE_LEVEL,RMM_RTT_TREE_PRIMARY).level)).state == ASSIGNED
  rtte_ripas                   post: RttEntryAt(RttAt(RttWalk(RealmAt(rd), ipa,RMM_RTT_PAGE_LEVEL,RMM_RTT_TREE_PRIMARY).rtt_addr), RttEntryIndex(ipa, RttWalk(RealmAt(rd), ipa,RMM_RTT_PAGE_LEVEL,RMM_RTT_TREE_PRIMARY).level)).ripas == RAM
  rtte_out                     post: RttEntryAt(RttAt(RttWalk(RealmAt(rd), ipa,RMM_RTT_PAGE_LEVEL,RMM_RTT_TREE_PRIMARY).rtt_addr), RttEntryIndex(ipa, RttWalk(RealmAt(rd), ipa,RMM_RTT_PAGE_LEVEL,RMM_RTT_TREE_PRIMARY).level)).out == data
  rtte_attr                    post: RttEntryAt(RttAt(RttWalk(RealmAt(rd), ipa,RMM_RTT_PAGE_LEVEL,RMM_RTT_TREE_PRIMARY).rtt_addr), RttEntryIndex(ipa, RttWalk(RealmAt(rd), ipa,RMM_RTT_PAGE_LEVEL,RMM_RTT_TREE_PRIMARY).level)).attr == MEMATTR_CACHEABLE
  rtte_sh                      post: RttEntryAt(RttAt(RttWalk(RealmAt(rd), ipa,RMM_RTT_PAGE_LEVEL,RMM_RTT_TREE_PRIMARY).rtt_addr), RttEntryIndex(ipa, RttWalk(RealmAt(rd), ipa,RMM_RTT_PAGE_LEVEL,RMM_RTT_TREE_PRIMARY).level)).sh == SHAREABILITY_INNER
  realm_rim                    post: RealmAt(rd).measurements[0] == RimExtendData(RealmAt(rd), data, flags)
