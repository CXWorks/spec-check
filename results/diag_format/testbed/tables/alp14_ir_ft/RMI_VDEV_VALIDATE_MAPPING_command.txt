B4.3.63 RMI_VDEV_VALIDATE_MAPPING command
                Completes a request made by the Realm to validate mappings to device memory from a target IPA range.
                B4.3.63.1 Interface
                B4.3.63.1.1   Input values
        Name                  Register    Bits        Type                                  Description
        fid                   X0          63:0        UInt64                                FID, value 0xC4000163
        rd                    X1          63:0        Address                               PA of the RD for the target Realm
        rec_ptr               X2          63:0        Address                               PA of the target REC
        pdev_ptr              X3          63:0        Address                               PA of the PDEV
        vdev_ptr              X4          63:0        Address                               PA of the VDEV
        base                  X5          63:0        Address                               Base of target IPA region
        top                   X6          63:0        Address                               Top of target IPA region
                B4.3.63.1.2   Context
                The RMI_VDEV_VALIDATE_MAPPING command operates on the following context.
        Name                  Type                       Value                                      Before   Description
        realm                 RmmRealm                   RealmAt(rd)                                false    Realm
        realm_pre             RmmRealm                   RealmAt(rd)                                true     Realm
        rec                   RmmRec                     RecAt(rec_ptr)                             false    REC
        pdev                  RmmPdev                    PdevAt(pdev_ptr)                           false    PDEV
        vdev                  RmmVdev                    VdevAt(vdev_ptr)                           false    VDEV
        pa_pre                Address                    rec.dev_mem_pa                             true     Output base address
        walk                  RmmRttWalkResult           RttWalk(                                   false    RTT walk result
                                              realm, base,
                                              RMM_RTT_PAGE_LEVEL,
                                              RMM_RTT_TREE_PRIMARY)
        walk_top_pre          Address                    RttSkipEntriesWithRipas(                   true     Top IPA of entries
                                              RttAt(walk.rtt_addr),                            which have associated
                                              walk.level,                                      RIPAS values,
                                              base, top,                                       starting from entry at
                                              FALSE)                                           which the RTT walk
                                                                                                             terminated
                B4.3.63.1.3   Output values
        Name                  Register    Bits        Type                                  Description
        result                X0          63:0        RmiCommandReturnCode                  Command return status
        out_top               X1          63:0        Address                               Top IPA of range whose RIPAS was
                                                                              modified
                B4.3.63.2 Failure conditions
  ID                           Condition
  rd_align                     pre: !AddrIsGranuleAligned(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_delegable                 pre: !PaIsDelegable(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_state                     pre: GranuleAt(rd).state != RD
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rec_ptr_align                pre: !AddrIsGranuleAligned(rec_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rec_ptr_delegable            pre: !PaIsDelegable(rec_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rec_ptr_state                pre: GranuleAt(rec_ptr).state != REC
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rec_state                    pre: RecAt(rec_ptr).state == REC_RUNNING
                               post: ResultEqual(result, RMI_ERROR_REC)
  rec_owner                    pre: RecAt(rec_ptr).owner != RealmAt(rd)
                               post: ResultEqual(result, RMI_ERROR_REC)
  pdev_ptr_align               pre: !AddrIsGranuleAligned(pdev_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  pdev_ptr_delegable           pre: !PaIsDelegable(pdev_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  pdev_ptr_state               pre: GranuleAt(pdev_ptr).state != PDEV
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  vdev_ptr_align               pre: !AddrIsGranuleAligned(vdev_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  vdev_ptr_delegable           pre: !PaIsDelegable(vdev_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  vdev_ptr_state               pre: GranuleAt(vdev_ptr).state != VDEV
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  vdev_pdev                    pre: VdevAt(vdev_ptr).pdev != pdev_ptr
                               post: ResultEqual(result, RMI_ERROR_DEVICE)
  top_le_base                  pre: top <= base
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  base_dev_mem_addr            pre: base != RecAt(rec_ptr).dev_mem_addr
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  top_gt_dev_mem_top           pre: top > RecAt(rec_ptr).dev_mem_top
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  base_rtt_align               pre: !AddrIsRttLevelAligned(base, RMM_RTT_PAGE_LEVEL)
                               post: ResultEqual(result, RMI_ERROR_RTT(0))
  top_align                    pre: !AddrIsGranuleAligned(top)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  base_walk_top                pre: base == RttSkipEntriesWithRipas(RttAt(walk.rtt_addr), walk.level, base, top, false)
                               post: ResultEqual(result, RMI_ERROR_RTT(0))
  dev_mem_non_coh_memattr      pre: RecAt(rec_ptr).dev_mem_flags.coh == DEV_MEM_NON_COHERENT && exists|rtte: RmmRttEntry| RttAt(walk.rtt_addr).entries[rtte].attr_prot != MEMATTR_NON_CACHEABLE
                               post: ResultEqual(result, RMI_ERROR_RTT(0))
  dev_mem_non_coh_map          pre: RecAt(rec_ptr).dev_mem_flags.coh == DEV_MEM_NON_COHERENT && exists|rtte: RmmRttEntry| RttAt(walk.rtt_addr).entries[rtte].sh != SHAREABILITY_OUTER
                               post: ResultEqual(result, RMI_ERROR_RTT(0))
  dev_mem_coh_memattr          pre: RecAt(rec_ptr).dev_mem_flags.coh == DEV_MEM_COHERENT && exists|rtte: RmmRttEntry| RttAt(walk.rtt_addr).entries[rtte].attr_prot != MEMATTR_PASSTHROUGH
                               post: ResultEqual(result, RMI_ERROR_RTT(0))
  dev_mem_coh_map              pre: RecAt(rec_ptr).dev_mem_flags.coh == DEV_MEM_COHERENT && exists|rtte: RmmRttEntry| RttAt(walk.rtt_addr).entries[rtte].sh == SHAREABILITY_INNER
                               post: ResultEqual(result, RMI_ERROR_RTT(0))
  out_top_contiguous           pre: !RttSkipEntriesWithRipas(RttAt(walk.rtt_addr), walk.level, RecAt(rec_ptr).dev_mem_pa, RttSkipEntriesWithRipas(RttAt(walk.rtt_addr), walk.level, base, top, false), false)
                               post: ResultEqual(result, RMI_ERROR_RTT(0))
  aux_rtt_live                 pre: exists|addr: Address| AddrInRange(Addr, addr, base, top - base + 1) && RealmIsLive(RealmAt(rd))
                               post: ResultEqual(result, RMI_ERROR_RTT(0))
                B4.3.63.2.1   Failure condition ordering
                [rd_delegable, rd_state] < [base_rtt_align]
                                 rd_delegable                     rd_state                        base_rtt_align
                                                                                           RMI_ERROR_INPUT
                                                                                           base_walk_top
                [rd_delegable, rd_state] < [base_walk_top]
                                 rd_delegable                     rd_state                        base_walk_top
                                                                                           RMI_ERROR_INPUT
                [rec_ptr_delegable, rec_ptr_state] < [rec_state, rec_owner]
                                 rec_ptr_delegable            rec_ptr_state                  rec_state                        rec_owner
                                                                                           RMI_ERROR_INPUT
                                                                                           RMI_ERROR_REC
                [pdev_ptr_delegable, pdev_ptr_state, vdev_ptr_delegable, vdev_ptr_state] < [vdev_pdev]
                                 pdev_ptr_delegable            pdev_ptr_state                  vdev_ptr_delegable            vdev_ptr_state                  vdev_pdev
                                                                                           RMI_ERROR_INPUT
                                                                                           RMI_ERROR_INPUT
                                                                                           RMI_ERROR_INPUT
                                                                                           RMI_ERROR_DEVICE
                [base_dev_mem_addr] < [base_rtt_align]
                                 base_dev_mem_addr            base_rtt_align
                                                                                           RMI_ERROR_INPUT
                                                                                           RMI_ERROR_RTT(0)
                [top_align] < [base_walk_top]
                                 top_align                     base_walk_top
                                                                                           RMI_ERROR_INPUT
                                                                                           RMI_ERROR_RTT(0)
                B4.3.63.3 Success conditions
  rec_ripas                   post: RttAt(walk.rtt_addr).entries[0].ripas == DEV
  dev_mem_addr                 post: RecAt(rec_ptr).dev_mem_addr == min(top, RttSkipEntriesWithRipas(RttAt(walk.rtt_addr), walk.level, base, top, false))
  dev_mem_pa                   post: RecAt(rec_ptr).dev_mem_pa == RecAt(rec_ptr).dev_mem_pa + (RttSkipEntriesWithRipas(RttAt(walk.rtt_addr), walk.level, base, top, false) - base)
  out_top                      post: out_top == min(top, RttSkipEntriesWithRipas(RttAt(walk.rtt_addr), walk.level, base, top, false))
