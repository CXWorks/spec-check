B4.3.30 RMI_REC_ENTER command
              Enter a REC.
              B4.3.30.1 Interface
              B4.3.30.1.1   Input values
        Name                  Register    Bits        Type                                  Description
        fid                   X0          63:0        UInt64                                FID, value 0xC400015C
        rec_ptr              X1          63:0        Address                               PA of the target REC
        run_ptr              X2          63:0        Address                               PA of RecRun object
                B4.3.30.1.2   Context
                The RMI_REC_ENTER command operates on the following context.
        Name                  Type                       Value                                      Before   Description
        run                  RmiRecRun                  RmiRecRunAt(run_ptr)                       false    RecRun object
        rec                  RmmRec                     RecAt(rec_ptr)                             false    REC
        realm                RmmRealm                   RealmAt(RecAt(rec_ptr).owner)              false    Realm
                B4.3.30.1.3   Output values
        Name                  Register    Bits        Type                                  Description
        result                X0          63:0        RmiCommandReturnCode                  Command return status
                B4.3.30.2 Failure conditions
  ID                           Condition
  rec_ptr_align               pre: !AddrIsGranuleAligned(rec_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rec_ptr_delegable           pre: !PaIsDelegable(rec_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rec_ptr_state               pre: GranuleAt(rec_ptr).state != REC
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  run_ptr_align               pre: !AddrIsGranuleAligned(run_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  run_ptr_ns_access           pre: !GranuleAccessPermitted(RmiRecRunAt(run_ptr).enter.gicv3_hcr, PAS_NS)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rec_state                   pre: RecAt(rec_ptr).state == REC_RUNNING
                               post: ResultEqual(result, RMI_ERROR_REC)
  rec_runnable                pre: RecAt(rec_ptr).flags.runnable == NOT_RUNNABLE
                               post: ResultEqual(result, RMI_ERROR_REC)
  rec_pending                 pre: RecAt(rec_ptr).pending != REC_PENDING_NONE
                               post: ResultEqual(result, RMI_ERROR_REC)
  realm_state_new             pre: RealmAt(RecAt(rec_ptr).owner).state == REALM_NEW
                               post: ResultEqual(result, RMI_ERROR_REALM(0))
  realm_state_off             pre: RealmAt(RecAt(rec_ptr).owner).state == REALM_SYSTEM_OFF
                               post: ResultEqual(result, RMI_ERROR_REALM(1))
  run_emul_mmio               pre: RmiRecRunAt(run_ptr).enter.flags.emul_mmio == RMI_EMULATED_MMIO && RmiRecRunAt(run_ptr).enter.flags.emul_mmio == RMI_EMULATED_MMIO && RecAt(rec_ptr).emulatable_abort != EMULATABLE_ABORT
                               post: ResultEqual(result, RMI_ERROR_REC)
  run_gicv3_valid             pre: !Gicv3ConfigIsValid(RmiRecRunAt(run_ptr).enter.gicv3_hcr, RmiRecRunAt(run_ptr).enter.gicv3_lrs)
                               post: ResultEqual(result, RMI_ERROR_REC)
                B4.3.30.2.1   Failure condition ordering
                [rec_ptr_align, rec_ptr_delegable, rec_ptr_state, run_ptr_align, run_ptr_ns_access] < [rec_state, rec_runnable, rec_pending, realm_state_new, realm_state_off, run_emul_mmio, run_gicv3_valid]
                                                                                           rec_ptr_align                     rec_ptr_delegable                        rec_ptr_state
                                                                                                                           run_ptr_align                     run_ptr_ns_access
                                                                                           rec_state                        rec_runnable                        rec_pending
                                                                                                                           realm_state_new                     realm_state_off
                                                                                                                           run_emul_mmio                        run_gicv3_valid
                                                                                           RMI_ERROR_INPUT
                                                                                           RMI_ERROR_REC
                                                                                           realm_state
                B4.3.30.3 Success conditions
  rec_emulatable_abort        post: RecAt(rec_ptr).emulatable_abort == EMULATABLE_ABORT
                B4.3.30.4 Footprint
    ID                        Value
    rec_emulatable_abort      RecAt(rec_ptr).emulatable_abort
