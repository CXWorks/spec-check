B4.3.25 RMI_REALM_CREATE command
              Creates a Realm.
              B4.3.25.1 Interface
              B4.3.25.1.1   Input values
        Name                  Register    Bits        Type                                  Description
        fid                   X0          63:0        UInt64                                FID, value 0xC4000158
        rd                    X1          63:0        Address                               PA of the RD
        params_ptr            X2          63:0        Address                               PA of Realm parameters
                B4.3.25.1.2   Context
                The RMI_REALM_CREATE command operates on the following context.
        Name                  Type                       Value                                      Before   Description
        params                RmiRealmParams             RmiRealmParamsAt(                          false    Realm parameters
                                                           params_ptr)
        realm                 RmmRealm                   RealmAt(rd)                                false    Realm
        mec_members_pre       UInt64                         MecMembers(params.mecid)                   true     Number of Realms
                                                                                                           which are members of
                                                                                                           the Realm's MEC
        mec_state_pre         RmmMecState                MecState(params.mecid)                     true     MECID state
                B4.3.25.1.3   Output values
        Name                  Register    Bits        Type                                  Description
        result                X0          63:0        RmiCommandReturnCode                  Command return status
                B4.3.25.2 Failure conditions
  ID                           Condition
  params_ptr_align            pre: !AddrIsGranuleAligned(params_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  params_ptr_access           pre: !AddrIsProtected(params_ptr, CurrentRealm())
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  params_ptr_valid            pre: !RmiRealmParamsIsValid(params_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  params_supported            pre: !RealmParamsSupported(RmiRealmParamsAt(params_ptr))
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_rtt_base                 pre: AddrInRange(rd, RmiRealmParamsAt(params_ptr).rtt_base, RmiRealmParamsAt(params_ptr).rtt_num_start - 1)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_align                    pre: !AddrIsGranuleAligned(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_delegable                pre: !PaIsDelegable(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_state                    pre: GranuleAt(rd).state != DELEGATED
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rtt_base_align              pre: !AddrIsGranuleAligned(RmiRealmParamsAt(params_ptr).rtt_base)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rtt_config                  pre: !(RmiRealmParamsAt(params_ptr).s2sz && RmiRealmParamsAt(params_ptr).rtt_level_start && RmiRealmParamsAt(params_ptr).rtt_num_start)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rtt_state                   pre: !GranulesAllState(RmiRealmParamsAt(params_ptr).rtt_base, RmiRealmParamsAt(params_ptr).rtt_num_start - 1, DELEGATED)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  vmid_valid                  pre: !VmidsAreValid(RmiRealmParamsAt(params_ptr).vmid, RmiRealmParamsAt(params_ptr).aux_vmid)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  mecid_max                   pre: params.mecid > 0
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  mec_state                   pre: MecState(params.mecid) == MEC_STATE_PRIVATE_ASSIGNED
                               post: ResultEqual(result, RMI_ERROR_INPUT)
