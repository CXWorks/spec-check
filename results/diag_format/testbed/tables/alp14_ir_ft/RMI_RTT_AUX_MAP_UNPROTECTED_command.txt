B4.3.35 RMI_RTT_AUX_MAP_UNPROTECTED command
              Copies a valid or table descriptor at an Unprotected IPA from the starting level of the primary RTT to an auxiliary
              RTT.
              B4.3.35.1 Interface
              B4.3.35.1.1   Input values
        Name                  Register    Bits        Type                                  Description
        fid                   X0          63:0        UInt64                                FID, value 0xC4000181
        rd                    X1          63:0        Address                               PA of the RD for the target Realm
        ipa                   X2          63:0        Address                               IPA in the target Realm
        index                 X3          63:0        UInt64                                RTT tree index
                B4.3.35.1.2   Context
                The RMI_RTT_AUX_MAP_UNPROTECTED command operates on the following context.
        Name                  Type                       Value                                      Before   Description
        realm                 RmmRealm                   RealmAt(rd)                                false    Realm
        walk_pri              RmmRttWalkResult            RttWalk(                                   false    Primary RTT walk
                                  realm, ipa,                                     result
                                  realm.rtt_level_start,
                                  RMM_RTT_TREE_PRIMARY)
        walk_aux              RmmRttWalkResult            RttWalk(                                   false    Auxiliary RTT walk
                                  realm, ipa,                                     result
                                  realm.rtt_level_start,
                                  index)
        entry_idx             UInt64                      RttEntryIndex(                             false    RTTE index
                                  ipa, walk_aux.level)
                B4.3.35.1.3   Output values
        Name                  Register    Bits        Type                                  Description
        result                X0          63:0        RmiCommandReturnCode                  Command return status
                B4.3.35.2 Failure conditions
  ID                           Condition
  rd_align                     pre: !AddrIsGranuleAligned(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_delegable                 pre: !PaIsDelegable(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_state                     pre: GranuleAt(rd).state != RD
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  ipa_aligned                  pre: !AddrIsRttLevelAligned(ipa, realm.rtt_level_start)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  ipa_in_range                 pre: !(ipa < pow2(realm.ipa_width))
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  ipa_protected                pre: AddrIsProtected(ipa, realm)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  realm_rtt_tree               pre: realm.rtt_tree_per_plane == FEATURE_FALSE
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  index_primary                pre: index == RMM_RTT_TREE_PRIMARY
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  index_too_high               pre: index > realm.num_aux_planes
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  walk_pri_unassigned          pre: walk_pri.rtte.state == UNASSIGNED_NS
                               post: ResultEqual(result, RMI_ERROR_RTT(RMM_RTT_TREE_PRIMARY))
                B4.3.35.2.1   Failure condition ordering
                [rd_delegable, rd_state] < [walk_pri_unassigned]
                                 rd_align                     rd_state                        rd_delegable
                                                                                           RMI_ERROR_INPUT
                                                                                           walk_pri_unassigned
                [ipa_aligned, ipa_in_range, ipa_protected] < [walk_pri_unassigned]
                                 ipa_aligned                  ipa_in_range                  ipa_protected
                                                                                           RMI_ERROR_INPUT
                                                                                           walk_pri_unassigned
                [realm_rtt_tree, index_primary, index_too_high] < [walk_pri_unassigned]
                                 realm_rtt_tree              index_primary                index_too_high
                                                                                           RMI_ERROR_INPUT
                                                                                           walk_pri_unassigned
                B4.3.35.3 Success conditions
  walk_aux_state               post: walk_aux.rtte.state == walk_pri.rtte.state
  walk_aux_attr_unprot         post: RttEntryStateToRmi(walk_aux.rtte, walk_aux.rtte.attr_unprot) == RttEntryStateToRmi(walk_pri.rtte, walk_pri.rtte.attr_unprot)
  walk_aux_s2ap                post: RttS2APEqual(walk_aux.rtte, walk_aux.rtte, walk_aux.rtte.s2ap_indirect.s2ap_indirect.encoding) == RttS2APEqual(walk_pri.rtte, walk_pri.rtte, walk_pri.rtte.s2ap_indirect.s2ap_indirect.encoding)
  walk_aux_addr                post: walk_aux.rtte.s2ap_indirect.s2ap_indirect.base_index == walk_pri.rtte.s2ap_indirect.s2ap_indirect.base_index
                B4.3.35.4 Footprint
    ID                           Value
    rtte                        RttEntry(walk_aux.rtt_addr, entry_idx)
