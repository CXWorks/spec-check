B4.3.42 RMI_RTT_MAP_UNPROTECTED command
                Creates a mapping from an Unprotected IPA to a Non-secure PA in a primary RTT.
                B4.3.42.1 Interface
                B4.3.42.1.1   Input values
        Name                  Register    Bits        Type                                  Description
        fid                   X0          63:0        UInt64                                FID, value 0xC400015F
        rd                    X1          63:0        Address                               PA of the RD for the target Realm
        ipa                   X2          63:0        Address                               IPA at which the memory will be mapped in the target Realm
        level                 X3          63:0        Int64                                 RTT level
        desc                  X4          63:0        Bits64                                RTTE descriptor
                B4.3.42.1.2   Context
                The RMI_RTT_MAP_UNPROTECTED command operates on the following context.
        Name                  Type                       Value                                      Before   Description
        realm                 RmmRealm                   RealmAt(rd)                                false    Realm
        walk                  RmmRttWalkResult           RttWalk(realm, ipa, level, RMM_RTT_TREE_PRIMARY)  false    RTT walk result
        entry_idx             UInt64                     RttEntryIndex(ipa, walk.level)             false    RTTE index
        rtte                  RmmRttEntry                RttDescriptorDecode(desc, realm.rtt_s2ap_encoding)  false    RTT entry
                B4.3.42.1.3   Output values
        Name                  Register    Bits        Type                                  Description
        result                X0          63:0        RmiCommandReturnCode                  Command return status
                B4.3.42.2 Failure conditions
  ID                           Condition
  rd_align                     pre: !AddrIsGranuleAligned(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_delegable                 pre: !PaIsDelegable(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_state                     pre: GranuleAt(rd).state != RD
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  level_valid                  pre: level < 1 || !RttLevelIsValid(realm, level)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  ipa_aligned                  pre: !AddrIsAligned(ipa, pow2(level))
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  ipa_range                    pre: ipa >= pow2(realm.ipa_width)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  ipa_protected                pre: AddrIsProtected(ipa, realm)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  desc_addr_aligned            pre: !AddrIsAligned(RttDescriptorDecode(desc, realm.rtt_s2ap_encoding).addr, pow2(level))
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  desc_addr_lpa2               pre: realm.feat_lpa2 == FEATURE_FALSE && RttDescriptorDecode(desc, realm.rtt_s2ap_encoding).addr >= pow2(48)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  desc_indirect_base           pre: realm.rtt_s2ap_encoding == S2AP_INDIRECT && !(RttDescriptorDecode(desc, realm.rtt_s2ap_encoding).indirect_base_index == S2AP_NO_ACCESS || RttDescriptorDecode(desc, realm.rtt_s2ap_encoding).indirect_base_index == S2AP_RO || RttDescriptorDecode(desc, realm.rtt_s2ap_encoding).indirect_base_index == S2AP_WO || RttDescriptorDecode(desc, realm.rtt_s2ap_encoding).indirect_base_index == S2AP_RW)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  walk_level                   pre: walk.level < level
                               post: ResultEqual(result, RMI_ERROR_RTT)
  walk_entry_state             pre: walk.rtte.state != UNASSIGNED_NS
                               post: ResultEqual(result, RMI_ERROR_RTT)
                B4.3.42.2.1   Failure condition ordering
                [rd_delegable, rd_state] < [level_valid, ipa_aligned, ipa_range, ipa_protected, desc_addr_aligned, desc_addr_lpa2, desc_indirect_base] < [walk_level, walk_entry_state]
  rd_delegable                 rd_state                        level_valid                     ipa_aligned                     ipa_range                     ipa_protected                     desc_addr_aligned                     desc_addr_lpa2                     desc_indirect_base
                                                                                           walk_level                     walk_entry_state
                B4.3.42.3 Success conditions
  walk_entry_state             post: walk.rtte.state == ASSIGNED_NS
  walk_unprotected_attrs       post: RttDescriptorDecode(walk.rtte, realm.rtt_s2ap_encoding).unprotected_attrs == RttDescriptorDecode(desc, realm.rtt_s2ap_encoding).unprotected_attrs
  walk_direct_perms            post: realm.rtt_s2ap_encoding == S2AP_DIRECT && (walk.rtte.read == RttDescriptorDecode(desc, realm.rtt_s2ap_encoding).read && walk.rtte.write == RttDescriptorDecode(desc, realm.rtt_s2ap_encoding).write)
  walk_indirect_base           post: realm.rtt_s2ap_encoding == S2AP_INDIRECT && walk.rtte.indirect_base_index == RttDescriptorDecode(desc, realm.rtt_s2ap_encoding).indirect_base_index && walk.rtte.overlay_index == 15
  walk_addr                    post: walk.rtte.addr == RttDescriptorDecode(desc, realm.rtt_s2ap_encoding).addr
                B4.3.42.4 Footprint
    ID                           Value
    rtte                        RttEntryAt(RttAt(walk.rtt_addr), entry_idx)
