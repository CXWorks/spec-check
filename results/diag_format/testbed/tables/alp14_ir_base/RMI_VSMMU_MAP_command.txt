B4.3.67 RMI_VSMMU_MAP command
                Create a VSMMU mapping.
                B4.3.67.1 Interface
                B4.3.67.1.1   Input values
        Name                  Register    Bits        Type                                  Description
        fid                   X0          63:0        UInt64                                FID, value 0xC400016C
        rd                    X1          63:0        Address                               PA of the RD
        vsmmu_ptr             X2          63:0        Address                               PA of the VSMMU
        ipa                   X3          63:0        Address                               IPA at which to create the mapping
        level                 X4          63:0        Int64                                 RTT level
                B4.3.67.1.2   Context
                The RMI_VSMMU_MAP command operates on the following context.
        Name                  Type                       Value                                      Before   Description
        realm                 RmmRealm                   RealmAt(rd)                                false    Realm
        vsmmu                 RmmVsmmu                   VsmmuAt(vsmmu_ptr)                         false    VSMMU
        walk                  RmmRttWalkResult           RttWalk(                                   false    RTT walk result
                                                             realm, ipa, level,
                                                             RMM_RTT_TREE_PRIMARY)
        entry_idx             UInt64                     RttEntryIndex(                             false    RTTE index
                                                             ipa, walk.level)
                B4.3.67.1.3   Output values
        Name                  Register    Bits        Type                                  Description
        result                X0          63:0        RmiCommandReturnCode                  Command return status
                B4.3.67.2 Failure conditions
  ID                           Condition
  da_supported                pre: !ImplFeatures(CurrentRealm(s)).feat_da
                               post: ResultEqual(result, RMI_ERROR_NOT_SUPPORTED)
  rd_align                    pre: !AddrIsGranuleAligned(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_delegable                pre: !PaIsDelegable(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_state                    pre: GranuleAt(rd).state != RD
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  realm_state                 pre: realm.state != REALM_NEW
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  level_valid                 pre: !RttLevelIsValid(realm, level) || level < 2
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  vsmmu_ptr_align             pre: !AddrIsGranuleAligned(vsmmu_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  vsmmu_ptr_delegable         pre: !PaIsDelegable(vsmmu_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  vsmmu_ptr_state             pre: GranuleAt(vsmmu_ptr).state != VSMMU
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  ipa_align                   pre: !AddrIsAligned(ipa, pow2(level))
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  ipa_below_base              pre: ipa < VsmmuAt(vsmmu_ptr).reg_base
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  walk_level                  pre: walk.level < level
                               post: ResultEqual(result, RMI_ERROR_RTT)
  walk_entry_state            pre: walk.rtte.state != UNASSIGNED
                               post: ResultEqual(result, RMI_ERROR_RTT)
  walk_entry_ripas            pre: walk.rtte.ripas != EMPTY
                               post: ResultEqual(result, RMI_ERROR_RTT)
  walk_range_top              pre: ipa + pow2(walk.level) >= VsmmuAt(vsmmu_ptr).reg_top
                               post: ResultEqual(result, RMI_ERROR_RTT)
                B4.3.67.2.1   Failure condition ordering
                [da_supported, rd_align, rd_delegable, rd_state, realm_state] < [level_valid, vsmmu_ptr_align, vsmmu_ptr_delegable, vsmmu_ptr_state] < [ipa_align, ipa_below_base] < [walk_level, walk_entry_state, walk_entry_ripas, walk_range_top]
                                                                                           da_supported
                                                                                           rd_align
                                                                                           rd_delegable
                                                                                           rd_state
                                                                                           realm_state
                                                                                           level_valid
                                                                                           vsmmu_ptr_align
                                                                                           vsmmu_ptr_delegable
                                                                                           vsmmu_ptr_state
                                                                                           ipa_align
                                                                                           ipa_below_base
                                                                                           walk_level
                                                                                           walk_entry_state
                                                                                           walk_entry_ripas
                                                                                           walk_range_top
                B4.3.67.3 Success conditions
  ID                           Condition
  rtte_state                  post: RttEntryAt(RttAt(walk.rtte.rtt_addr), entry_idx).state == ASSIGNED_VSMMU
  rtte_addr                   post: RttEntryAt(RttAt(walk.rtte.rtt_addr), entry_idx).addr == vsmmu_ptr
                B4.3.67.4 Footprint
    ID                        Value
    rtte                      RttEntryAt(RttAt(walk.rtte.rtt_addr), entry_idx)
