B4.3.38 RMI_RTT_CREATE command
                Creates a primary RTT.
                B4.3.38.1 Interface
                B4.3.38.1.1   Input values
        Name                  Register    Bits        Type                                  Description
        fid                   X0          63:0        UInt64                                FID, value 0xC400015D
        rd                    X1          63:0        Address                               PA of the RD for the target Realm
        rtt                   X2          63:0        Address                               PA of the target RTT
        ipa                   X3          63:0        Address                               Base of the IPA range described by the RTT
        level                 X4          63:0        Int64                                 RTT level
                B4.3.38.1.2   Context
                The RMI_RTT_CREATE command operates on the following context.
        Name                  Type                       Value                                      Before   Description
        realm                 RmmRealm                   RealmAt(rd)                                false    Realm
        walk                  RmmRttWalkResult           RttWalk(realm, ipa, level - 1, RMM_RTT_TREE_PRIMARY)  false    RTT walk result
        entry_idx             UInt64                     RttEntryIndex(ipa, walk.level)             false    RTTE index
        walk_pre              RmmRttWalkResult           RttWalk(realm, ipa, level - 1, RMM_RTT_TREE_PRIMARY)  true     RTT walk result before command execution
        rtte_pre              RmmRttEntry                walk_pre.rtte                              true     RTTE before command execution
                B4.3.38.1.3   Output values
        Name                  Register    Bits        Type                                  Description
        result                X0          63:0        RmiCommandReturnCode                  Command return status
                B4.3.38.2 Failure conditions
  ID                           Condition
  rd_align                     pre: !AddrIsGranuleAligned(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_delegable                 pre: !PaIsDelegable(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_state                     pre: GranuleAt(rd).state != RD
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  level_valid                  pre: !RttLevelIsValid(realm, level) || level == RttLevelIsStarting(realm, level)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  ipa_aligned                  pre: !AddrIsAligned(ipa, pow2(RttLevelSize(level - 1)))
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  ipa_in_range                 pre: !AddrIsWithin(ipa, 0, pow2(realm.ipa_width))
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rtt_align                    pre: !AddrIsGranuleAligned(rtt)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rtt_delegable_dram           pre: !PaIsDelegableDram(rtt)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rtt_state                    pre: GranuleAt(rtt).state != DELEGATED
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  lpa2_check                   pre: realm.feat_lpa2 == FEATURE_FALSE && rtt >= pow2(48)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  walk_below                   pre: walk_pre.level < level - 1
                               post: ResultEqual(result, RMI_ERROR_RTT)
  rtte_table                   pre: rtte_pre.state == TABLE
                               post: ResultEqual(result, RMI_ERROR_RTT)
                B4.3.38.2.1   Failure condition ordering
                [rd_delegable, rd_state, level_valid, ipa_in_range] < [lpa2_check, rtt_align, rtt_delegable_dram, rtt_state, walk_below, rtte_table]
  rd_delegable                 rd_state                        level_valid                     ipa_in_range                        lpa2_check
  rd_state                     level_valid                     ipa_in_range                        lpa2_check                        rtt_align
  level_valid                     ipa_in_range                        lpa2_check                        rtt_align                        rtt_delegable_dram
  ipa_in_range                        lpa2_check                        rtt_align                        rtt_delegable_dram                        rtt_state
  lpa2_check                        rtt_align                        rtt_delegable_dram                        rtt_state                        walk_below
  rtt_align                        rtt_delegable_dram                        rtt_state                        walk_below                        rtte_table
  rtt_delegable_dram                        rtt_state                        walk_below                        rtte_table
  rtt_state                        walk_below                        rtte_table
  walk_below                        rtte_table
                B4.3.38.3 Success conditions
  rtt_state                    post: GranuleAt(rtt).state == RTT
  rtte_table                   post: RttEntryAt(RttAt(walk.rtt_addr), entry_idx).state == TABLE
  rtte_rtt                     post: RttEntryAt(RttAt(walk.rtt_addr), entry_idx).rtte == rtt
  ripas_preserved              post: forall|entry: RmmRttEntry| entry.state == TABLE ==> entry.ripas == rtte_pre.ripas
  state_preserved              post: forall|entry: RmmRttEntry| entry.state == TABLE ==> entry.state == rtte_pre.state
  contiguous_output            post: (forall|entry: RmmRttEntry| entry.state == TABLE ==> entry.ripas == rtte_pre.ripas) && (forall|entry: RmmRttEntry| entry.state == TABLE ==> entry.state == rtte_pre.state) && (rtte_pre.state != UNASSIGNED && rtte_pre.state != UNASSIGNED_NS) ==> (forall|i: 0..RttLevelSize(level)| AddrIsWithin(rtte_pre.ripas + i * pow2(RttLevelSize(level)), rtte_pre.ripas, rtte_pre.ripas + RttLevelSize(level) * pow2(RttLevelSize(level))))
                B4.3.38.4 Footprint
    ID                        Value
    rtt_state                 GranuleAt(rtt).state
    rtte                      RttEntryAt(RttAt(walk.rtt_addr), entry_idx)
