B4.3.34 RMI_RTT_AUX_MAP_PROTECTED command
                Creates a mapping from an Protected IPA in an auxiliary RTT.
                B4.3.34.1 Interface
                B4.3.34.1.1   Input values
        Name                  Register    Bits        Type                                  Description
        fid                   X0          63:0        UInt64                                FID, value 0xC4000180
        rd                    X1          63:0        Address                               PA of the RD for the target Realm
        ipa                   X2          63:0        Address                               IPA in the target Realm
        index                 X3          63:0        UInt64                                RTT tree index
                B4.3.34.1.2   Context
                The RMI_RTT_AUX_MAP_PROTECTED command operates on the following context.
        Name                  Type                       Value                                      Before   Description
        realm                 RmmRealm                   RealmAt(rd)                                false    Realm
        walk_pri              RmmRttWalkResult           RttWalk(                                   false    Primary RTT walk
                                                             realm, ipa,                                     result
                                                             RMM_RTT_PAGE_LEVEL,
                                                             RMM_RTT_TREE_PRIMARY)
        walk_aux              RmmRttWalkResult           RttWalk(                                   false    Auxiliary RTT walk
                                                             realm, ipa,                                     result
                                                             RMM_RTT_PAGE_LEVEL,
                                                             index)
        entry_idx             UInt64                     RttEntryIndex(                             false    RTTE index
                                                             ipa, walk_aux.level)
                B4.3.34.1.3   Output values
        Name                  Register    Bits        Type                                  Description
        result                X0          63:0        RmiCommandReturnCode                  Command return status
        state                 X1          7:0         RmiRttEntryState                      State of RTT entry whose contents
                                                                                            caused command to fail with
                                                                                            RMI_ERROR_RTT
        ripas                 X2          7:0         RmiRipas                              RIPAS of RTT entry which caused
                                                                                            command to fail with
                                                                                            RMI_ERROR_RTT
                The following unused bits of RMI_RTT_AUX_MAP_PROTECTED output values MBZ: X1[63:8], X2[63:8].
              B4.3.34.2 Failure conditions
  ID                           Condition
  rd_align                     pre: !AddrIsGranuleAligned(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_delegable                 pre: !PaIsDelegable(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_state                     pre: GranuleAt(rd).state != RD
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  ipa_align                    pre: !AddrIsGranuleAligned(ipa)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  ipa_protected                pre: !AddrIsProtected(ipa, realm)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  realm_rtt_tree_per_plane     pre: realm.rtt_tree_per_plane == FEATURE_FALSE || index == RMM_RTT_TREE_PRIMARY || index > realm.num_aux_planes
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  walk_pri_state               pre: walk_pri.state != ASSIGNED && walk_pri.state != ASSIGNED_DEV && walk_pri.state != ASSIGNED_VSMMU
                               post: ResultEqual(result, RMI_ERROR_RTT)
  walk_pri_state_assigned_ram  pre: walk_pri.state == ASSIGNED && walk_pri.rtte.ripas != RAM
                               post: ResultEqual(result, RMI_ERROR_RTT)
  walk_pri_state_assigned_dev  pre: walk_pri.state == ASSIGNED_DEV && walk_pri.rtte.ripas != DEV
                               post: ResultEqual(result, RMI_ERROR_RTT)
  walk_aux_state               pre: walk_aux.state == AUX_DESTROYED
                               post: ResultEqual(result, RMI_ERROR_RTT_AUX)
  walk_aux_level               pre: walk_aux.level < walk_pri.level
                               post: ResultEqual(result, RMI_ERROR_RTT_AUX)
                B4.3.34.2.1   Failure condition ordering
                [rd_delegable, rd_state, ipa_protected, realm_rtt_tree_per_plane] < [walk_pri_state, walk_pri_state_assigned_ram, walk_pri_state_assigned_dev, walk_aux_state, walk_aux_level]
                                 rd_align                     rd_delegable                        rd_state                        ipa_align                     ipa_protected                     realm_rtt_tree_per_plane
                                                                                           walk_pri_state                        walk_pri_state_assigned_ram     walk_pri_state_assigned_dev     walk_aux_state                        walk_aux_level
                                                                                           RMI_ERROR_INPUT
                                                                                           RMI_ERROR_INPUT
                                                                                           RMI_ERROR_INPUT
                                                                                           RMI_ERROR_INPUT
                                                                                           RMI_ERROR_INPUT
                                                                                           RMI_ERROR_RTT
                                                                                           RMI_ERROR_RTT
                                                                                           RMI_ERROR_RTT
                                                                                           RMI_ERROR_RTT_AUX
                                                                                           RMI_ERROR_RTT_AUX
                B4.3.34.3 Success conditions
  ID                           Condition
  walk_aux_state               post: walk_aux.state == ASSIGNED
  walk_aux_attr_prot           post: walk_aux.rtte.attr_prot == walk_pri.rtte.attr_prot
  walk_aux_sh                  post: walk_aux.rtte.sh == walk_pri.rtte.sh
  walk_aux_addr                post: walk_aux.rtte.addr == walk_pri.rtte.addr + (entry_idx * pow2(walk_aux.level))
                B4.3.34.4 Footprint
    ID                           Value
    rtte                          RttEntry(walk_aux.rtt_addr, entry_idx)
