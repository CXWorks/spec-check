B4.3.63 RMI_VDEV_VALIDATE_MAPPING command
                Completes a request made by the Realm to validate mappings to device memory from a target IPA range.
                B4.3.63.1 Interface
                B4.3.63.1.1   Input values
        Name                  Register    Bits        Type                                  Description
        fid                   X0          63:0        UInt64                                FID, value 0xC4000163
        rd                    X1          63:0        Address                               PA of the RD for the target Realm
        rec_ptr               X2          63:0        Address                               PA of the target REC
        pdev_ptr              X3          63:0        Address                               PA of the PDEV
        vdev_ptr              X4          63:0        Address                               PA of the VDEV
        base                  X5          63:0        Address                               Base of target IPA region
        top                   X6          63:0        Address                               Top of target IPA region
                B4.3.63.1.2   Context
                The RMI_VDEV_VALIDATE_MAPPING command operates on the following context.
        Name                  Type                       Value                                      Before   Description
        realm                 RmmRealm                   RealmAt(rd)                                false    Realm
        realm_pre             RmmRealm                   RealmAt(rd)                                true     Realm
        rec                   RmmRec                     RecAt(rec_ptr)                             false    REC
        pdev                  RmmPdev                    PdevAt(pdev_ptr)                           false    PDEV
        vdev                  RmmVdev                    VdevAt(vdev_ptr)                           false    VDEV
        pa_pre                Address                    rec.dev_mem_pa                             true     Output base address
        walk                  RmmRttWalkResult           RttWalk(realm, base, RMM_RTT_PAGE_LEVEL, RMM_RTT_TREE_PRIMARY)  false    RTT walk result
        walk_top_pre          Address                    RttSkipEntriesWithRipas(RttAt(walk.rtt_addr), walk.level, base, top, FALSE)  true     Top IPA of entries which have associated RIPAS values, starting from entry at which the RTT walk terminated
                B4.3.63.1.3   Output values
        Name                  Register    Bits        Type                                  Description
        result                X0          63:0        RmiCommandReturnCode                  Command return status
        out_top               X1          63:0        Address                               Top IPA of range whose RIPAS was modified
                The out_top output value is valid only when the command result is RMI_SUCCESS.
                B4.3.63.2 Failure conditions
  ID                           Condition
  rd_align                     pre: !AddrIsGranuleAligned(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_delegable                 pre: !PaIsDelegable(rd)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rd_state                     pre: GranuleAt(rd).state != RD
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rec_align                    pre: !AddrIsGranuleAligned(rec_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rec_delegable                pre: !PaIsDelegable(rec_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rec_state                    pre: GranuleAt(rec_ptr).state != REC
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  rec_running                  pre: rec.state == REC_RUNNING
                               post: ResultEqual(result, RMI_ERROR_REC)
  rec_owner                    pre: !rec.owner == realm
                               post: ResultEqual(result, RMI_ERROR_REC)
  pdev_align                   pre: !AddrIsGranuleAligned(pdev_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  pdev_delegable               pre: !PaIsDelegable(pdev_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  pdev_state                   pre: GranuleAt(pdev_ptr).state != PDEV
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  vdev_align                   pre: !AddrIsGranuleAligned(vdev_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  vdev_delegable               pre: !PaIsDelegable(vdev_ptr)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  vdev_state                   pre: GranuleAt(vdev_ptr).state != VDEV
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  vdev_pdev_association        pre: !VdevAt(vdev_ptr).pdev == pdev_ptr
                               post: ResultEqual(result, RMI_ERROR_DEVICE)
  base_leq_top                 pre: top <= base
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  base_dev_mem_addr            pre: base != rec.dev_mem_addr
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  base_gt_dev_mem_top          pre: top > rec.dev_mem_top
                               post: ResultEqual(result, RMI_ERROR_INPUT)
  base_rtt_align               pre: !AddrIsAligned(base, RttLevelSize(walk.level))
                               post: ResultEqual(result, RMI_ERROR_RTT)
  top_gt_base                  pre: base == walk_top_pre
                               post: ResultEqual(result, RMI_ERROR_RTT)
  coherency_noncoherent_attr   pre: rec.dev_mem_flags.coh == DEV_MEM_NON_COHERENT && !RttEntriesInRangeMemAttr(RttAt(walk.rtt_addr), walk.level, base, walk_top_pre, MEMATTR_NON_CACHEABLE)
                               post: ResultEqual(result, RMI_ERROR_RTT)
  coherency_noncoherent_map    pre: rec.dev_mem_flags.coh == DEV_MEM_NON_COHERENT && !RttEntriesInRangeNonCohDevMem(RttAt(walk.rtt_addr), walk.level, base, walk_top_pre)
                               post: ResultEqual(result, RMI_ERROR_RTT)
  coherency_coherent_attr      pre: rec.dev_mem_flags.coh == DEV_MEM_COHERENT && !RttEntriesInRangeMemAttr(RttAt(walk.rtt_addr), walk.level, base, walk_top_pre, MEMATTR_PASSTHROUGH)
                               post: ResultEqual(result, RMI_ERROR_RTT)
  coherency_coherent_map       pre: rec.dev_mem_flags.coh == DEV_MEM_COHERENT && !RttEntriesInRangeCohDevMem(RttAt(walk.rtt_addr), walk.level, base, walk_top_pre)
                               post: ResultEqual(result, RMI_ERROR_RTT)
  out_top_contiguous           pre: !RttEntriesInRangeOutputContiguous(RttAt(walk.rtt_addr), walk.level, base, walk_top_pre, rec.dev_mem_pa)
                               post: ResultEqual(result, RMI_ERROR_RTT)
  aux_live                     pre: !RttAllEntriesState(RttAt(walk.rtt_addr), walk.level, base, walk_top_pre, UNASSIGNED)
                               post: ResultEqual(result, RMI_ERROR_RTT)
  top_align                    pre: !AddrIsGranuleAligned(top)
                               post: ResultEqual(result, RMI_ERROR_INPUT)
                B4.3.63.2.1   Failure condition ordering
                [rd_delegable, rd_state] < [base_rtt_align, top_gt_base]
                [rec_delegable, rec_state] < [rec_running, rec_owner]
                [pdev_delegable, pdev_state, vdev_delegable, vdev_state] < [vdev_pdev_association]
                [base_dev_mem_addr] < [base_rtt_align]
                [top_align] < [top_gt_base]
                B4.3.63.3 Success conditions
  ID                           Condition
  rtte_dev                     post: forall|rtte: RmmRttEntry| RttEntryAt(RttAt(walk.rtt_addr), walk.level, base, walk_top_pre, rtte).state == DEV
  dev_mem_addr                 post: rec.dev_mem_addr == MinAddress(top, walk_top_pre)
  dev_mem_pa                   post: rec.dev_mem_pa == pa_pre + (walk_top_pre - base)
  out_top                      post: out_top == MinAddress(top, walk_top_pre)
                B4.3.63.4 Footprint
    ID                           Value
    rtte                        RttAt(walk.rtt_addr)
    dev_mem_addr                rec.dev_mem_addr
    dev_mem_pa                  rec.dev_mem_pa
