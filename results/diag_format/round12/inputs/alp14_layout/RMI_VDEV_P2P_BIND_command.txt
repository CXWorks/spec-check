B4.3.58 RMI_VDEV_P2P_BIND command
              Create a P2P binding between two VDEVs.
              B4.3.58.1 Interface
              B4.3.58.1.1   Input values
        Name                Register    Bits        Type                                  Description
        fid                 X0          63:0        UInt64                                FID, value 0xC40001D4
        stream_ptr          X1          63:0        Address                               PA of the P2P_STREAM object
        rd                  X2          63:0        Address                               PA of the RD
        rec_ptr             X3          63:0        Address                               PA of the target REC
        pdev_1_ptr          X4          63:0        Address                               PA of the first PDEV object
        pdev_2_ptr          X5          63:0        Address                               PA of the second PDEV object
        vdev_1_ptr          X6          63:0        Address                               PA of the first VDEV object
        vdev_2_ptr          X7          63:0        Address                               PA of the second VDEV object
              B4.3.58.1.2   Context
              The RMI_VDEV_P2P_BIND command operates on the following context.
        Name                Type                       Value                                      Before   Description
        realm               RmmRealm                   RealmAt(rd)                                false    Realm
        rec                 RmmRec                     RecAt(rec_ptr)                             false    REC
        stream              RmmP2PStream               P2PStreamAt(stream_ptr)                    false    P2P stream
        pdev_1              RmmPdev                    PdevAt(pdev_1_ptr)                         false    First PDEV
        pdev_2              RmmPdev                    PdevAt(pdev_2_ptr)                         false    Second PDEV
        vdev_1              RmmVdev                    VdevAt(vdev_1_ptr)                         false    First VDEV
        vdev_2              RmmVdev                    VdevAt(vdev_2_ptr)                         false    Second VDEV
              B4.3.58.1.3   Output values
        Name                Register    Bits        Type                                  Description
        result              X0          63:0        RmiCommandReturnCode                  Command return status
              B4.3.58.2 Failure conditions

(da_supp) If ImplFeatures().feat_da != FEATURE_TRUE, then ResultEqual(result, RMI_ERROR_NOT_SUPPORTED).

(rd_align) If !AddrIsGranuleAligned(rd), then ResultEqual(result, RMI_ERROR_INPUT).

(rd_bound) If !PaIsDelegable(rd), then ResultEqual(result, RMI_ERROR_INPUT).

(rd_state) If GranuleAt(rd).state != RD, then ResultEqual(result, RMI_ERROR_INPUT).

(rec_align) If !AddrIsGranuleAligned(rec_ptr), then ResultEqual(result, RMI_ERROR_INPUT).

(rec_bound) If !PaIsDelegable(rec_ptr), then ResultEqual(result, RMI_ERROR_INPUT).

(rec_gran_state) If GranuleAt(rec_ptr).state != REC, then ResultEqual(result, RMI_ERROR_INPUT).

(rec_state) If rec.state == REC_RUNNING, then ResultEqual(result, RMI_ERROR_REC).

(rec_owner) If rec.owner != rd, then ResultEqual(result, RMI_ERROR_REC).

(stream_align) If !AddrIsGranuleAligned(stream_ptr), then ResultEqual(result, RMI_ERROR_INPUT).

(stream_bound) If !PaIsDelegable(stream_ptr), then ResultEqual(result, RMI_ERROR_INPUT).

(stream_state) If GranuleAt(stream_ptr).state != P2P_STREAM, then ResultEqual(result, RMI_ERROR_INPUT).

(pdev_1_align) If !AddrIsGranuleAligned(pdev_1_ptr), then ResultEqual(result, RMI_ERROR_INPUT).

(pdev_1_bound) If !PaIsDelegable(pdev_1_ptr), then ResultEqual(result, RMI_ERROR_INPUT).

(pdev_1_gran_state) If GranuleAt(pdev_1_ptr).state != PDEV, then ResultEqual(result, RMI_ERROR_INPUT).

(pdev_1_stream) If (pdev_1.p2p_stream_valid != RMM_TRUE || pdev_1.p2p_stream != stream_ptr), then ResultEqual(result, RMI_ERROR_INPUT).

(pdev_2_align) If !AddrIsGranuleAligned(pdev_2_ptr), then ResultEqual(result, RMI_ERROR_INPUT).

(pdev_2_bound) If !PaIsDelegable(pdev_2_ptr), then ResultEqual(result, RMI_ERROR_INPUT).

(pdev_2_gran_state) If GranuleAt(pdev_2_ptr).state != PDEV, then ResultEqual(result, RMI_ERROR_INPUT).

(pdev_2_stream) If (pdev_2.p2p_stream_valid != RMM_TRUE || pdev_2.p2p_stream != stream_ptr), then ResultEqual(result, RMI_ERROR_INPUT).

(vdev_1_align) If !AddrIsGranuleAligned(vdev_1_ptr), then ResultEqual(result, RMI_ERROR_INPUT).

(vdev_1_bound) If !PaIsDelegable(vdev_1_ptr), then ResultEqual(result, RMI_ERROR_INPUT).

(vdev_1_gran_state) If GranuleAt(vdev_1_ptr).state != VDEV, then ResultEqual(result, RMI_ERROR_INPUT).

(vdev_1_realm) If vdev_1.realm != rd, then ResultEqual(result, RMI_ERROR_INPUT).

(vdev_1_pdev) If vdev_1.pdev != pdev_1_ptr, then ResultEqual(result, RMI_ERROR_INPUT).

(vdev_1_comm) If vdev_1.comm_state != DEV_COMM_IDLE, then ResultEqual(result, RMI_ERROR_DEVICE).

(vdev_1_attest_info) If !VdevAttestInfoEqual(vdev_1.attest_info, rec.vdev_attest_info_1), then ResultEqual(result, RMI_ERROR_DEVICE).

(vdev_1_p2p_bound) If vdev_1.p2p_bound != FEATURE_FALSE, then ResultEqual(result, RMI_ERROR_DEVICE).

(vdev_2_align) If !AddrIsGranuleAligned(vdev_2_ptr), then ResultEqual(result, RMI_ERROR_INPUT).

(vdev_2_bound) If !PaIsDelegable(vdev_2_ptr), then ResultEqual(result, RMI_ERROR_INPUT).

(vdev_2_gran_state) If GranuleAt(vdev_2_ptr).state != VDEV, then ResultEqual(result, RMI_ERROR_INPUT).

(vdev_2_realm) If vdev_2.realm != rd, then ResultEqual(result, RMI_ERROR_INPUT).

(vdev_2_pdev) If vdev_2.pdev != pdev_2_ptr, then ResultEqual(result, RMI_ERROR_INPUT).

(vdev_2_comm) If vdev_2.comm_state != DEV_COMM_IDLE, then ResultEqual(result, RMI_ERROR_DEVICE).

(vdev_2_attest_info) If !VdevAttestInfoEqual(vdev_2.attest_info, rec.vdev_attest_info_2), then ResultEqual(result, RMI_ERROR_DEVICE).

(vdev_2_p2p_bound) If vdev_2.p2p_bound != FEATURE_FALSE, then ResultEqual(result, RMI_ERROR_DEVICE).
