B3.46 PdevFlags function
             Get RmiPdevFlags value.
             func PdevFlags(
                 pdev : RmmPdev) => RmiPdevFlags
             begin
                 var flags : RmiPdevFlags;
                  case pdev.spdm of
                      when SPDM_FALSE => flags.spdm = RMI_SPDM_FALSE;
                      when SPDM_TRUE => flags.spdm = RMI_SPDM_TRUE;
                   end
                   case pdev.ide of
                       when IDE_FALSE => flags.ide = RMI_IDE_FALSE;
                       when IDE_TRUE => flags.ide = RMI_IDE_TRUE;
                   end
                   case pdev.coh of
                       when IOCOH => flags.coh = RMI_IOCOH;
                       when FCOH => flags.coh = RMI_FCOH;
                   end
                   return flags;
             end
