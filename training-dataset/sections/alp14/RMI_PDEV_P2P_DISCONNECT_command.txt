B4.3.18 RMI_PDEV_P2P_DISCONNECT command
              Destroy a P2P stream between two PDEVs.
              B4.3.18.1 Interface
              B4.3.18.1.1   Input values
        Name                Register    Bits        Type                                  Description
        fid                 X0          63:0        UInt64                                FID, value 0xC4000182
        stream_ptr          X1          63:0        Address                               PA of the P2P_STREAM object
        pdev_1_ptr          X2          63:0        Address                               PA of the first PDEV object
        pdev_2_ptr          X3          63:0        Address                               PA of the second PDEV object
              B4.3.18.1.2   Context
              The RMI_PDEV_P2P_DISCONNECT command operates on the following context.
        Name                Type                       Value                                      Before   Description
        stream              RmmP2PStream               P2PStreamAt(stream_ptr)                    true     P2P stream
        pdev_1              RmmPdev                    PdevAt(pdev_1_ptr)                         false    First PDEV object
        pdev_2              RmmPdev                    PdevAt(pdev_2_ptr)                         false    Second PDEV object
              B4.3.18.1.3   Output values
        Name                Register    Bits        Type                                  Description
        result              X0          63:0        RmiCommandReturnCode                  Command return status
              B4.3.18.2 Failure conditions
  ID                         Condition
  da_supp                    pre: ImplFeatures().feat_da != FEATURE_TRUE
                             post: ResultEqual(result, RMI_ERROR_NOT_SUPPORTED)
  stream_align               pre: !AddrIsGranuleAligned(stream_ptr)
                             post: ResultEqual(result, RMI_ERROR_INPUT)
  stream_bound               pre: !PaIsDelegable(stream_ptr)
                             post: ResultEqual(result, RMI_ERROR_INPUT)
  stream_state               pre: GranuleAt(stream_ptr).state != P2P_STREAM
                             post: ResultEqual(result, RMI_ERROR_INPUT)
  ID                                Condition
  pdev_1_align                      pre: !AddrIsGranuleAligned(pdev_1_ptr)
                                    post: ResultEqual(result, RMI_ERROR_INPUT)
  pdev_1_bound                      pre: !PaIsDelegable(pdev_1_ptr)
                                    post: ResultEqual(result, RMI_ERROR_INPUT)
  pdev_1_gran_sta te                pre: GranuleAt(pdev_1_ptr).state != PDEV
                                    post: ResultEqual(result, RMI_ERROR_INPUT)
  pdev_2_align                      pre: !AddrIsGranuleAligned(pdev_2_ptr)
                                    post: ResultEqual(result, RMI_ERROR_INPUT)
  pdev_2_bound                      pre: !PaIsDelegable(pdev_2_ptr)
                                    post: ResultEqual(result, RMI_ERROR_INPUT)
  pdev_2_gran_sta te                pre: GranuleAt(pdev_2_ptr).state != PDEV
                                    post: ResultEqual(result, RMI_ERROR_INPUT)
  stream                            pre:  (pdev_1.p2p_stream_valid != RMM_TRUE
                                              || pdev_1.p2p_stream != stream_ptr
                                              || pdev_2.p2p_stream_valid != RMM_TRUE
                                              || pdev_2.p2p_stream != stream_ptr)
                                    post: ResultEqual(result, RMI_ERROR_INPUT)
               B4.3.18.2.1        Failure condition ordering
               [da_supp] < [stream_align, stream_bound, stream_state]
                                                                                                                                                                   da_supp
                                                                                                                                                              RMI_ERROR_NOT _SUPPORT ED
                   stream   pdev_2_gran_state   pdev_2_bound   pdev_2_align   pdev_1_gran_state   pdev_1_bound   pdev_1_align   stream_state   stream_bound      stream_align
                                                                                                                                                                    RMI_ERROR_INPUT
               B4.3.18.3 Success conditions
  ID                                Condition
  gran_state                        post: GranuleAt(stream_ptr).state == DELEGATED
  pdev_1_p2p_stream_valid post: pdev_1.p2p_stream_valid == RMM_FALSE
  pdev_1_state                      post: pdev_1.state == PDEV_COMMUNICATING
  pdev_1_comm_state                 post: pdev_1.comm_state == DEV_COMM_PENDING
  pdev_2_p2p_stream_valid post: pdev_2.p2p_stream_valid == RMM_FALSE
  pdev_2_state                      post: pdev_2.state == PDEV_COMMUNICATING
  pdev_2_comm_state                 post: pdev_2.comm_state == DEV_COMM_PENDING
